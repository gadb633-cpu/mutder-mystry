בודק למי שייך כלי נשק שדוקר והיה בחדר 8 בשעות שהקורבן היה שם.4
SELECT *
FROM(SELECT *
	FROM objects 
	WHERE isWeapon = "1" and (weaponType = "Stabbing" or weaponType = "Piercing")) as Stabbing_weapon
JOIN people_belongings as pb
ON Stabbing_weapon.objectId = pb.objectId
JOIN visits as v
ON v.personId = pb.personId 
WHERE (v.personid = "13" or v.personid = "25" or v.personid = "41" or v.personid = "50") AND v.roomId = "8"
GROUP by v.personid;
